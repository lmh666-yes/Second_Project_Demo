## 一、怎么放代码（不会动你的 main.c）

三选一：

1. **推荐**：新建 `_验收_并发与丢字节.c` 放进工程根目录，然后在 Keil 里 `Add Existing Files to Group` 加进 **Application/User** 组（或你放 FWLIB 源码的那个组），别的什么都不用改；
2. 直接把你手动创建的 `main.c` 里，在 `xTaskCreate` 那一段附近依次 `xTaskCreate` 下面几个任务；
3. 只跑单项时，把不需要的任务注释掉。

公共约定：

- 日志统一走 `SYS_USART_1`（`printf()` 已重定向到它，115200 8N1）；
- 诊断任务**栈给 256 字（1KB）**、优先级给 **2**（比采集/转发低，别抢业务）；
- 每个测试跑完就把任务注释掉，避免互相干扰；
- 里面所有 `printf` 格式串都是 ASCII（AC5 的 #870-D 红线），中文只在注释里。

## 二、T1 — A1：串口接收丢字节能查出来（约 5 分钟）

**目的**：证明"缓冲满丢字节"不再静默，并且把该路缓冲改大后确实不丢了。

```c
/* T1：不读接收缓冲，看丢字节计数 */
static void vTestUsartDrop(void *pv)
{
    (void)pv;
    SYS_USART_InitRxIT(SYS_USART_2, 115200);   /* 上位机那一路 */

    for (;;)
    {
        uint16_t drop, full, ovr, left;

        SYS_USART_RxDropReset(SYS_USART_2);
        printf("[T1] reset done, please send 1024 bytes within 5s...\r\n");
        vTaskDelay(pdMS_TO_TICKS(5000));       /* 这 5 秒里故意一个字节都不读 */

        drop = SYS_USART_RxDropCount(SYS_USART_2);
        full = SYS_USART_RxFullCount(SYS_USART_2);
        ovr  = SYS_USART_RxOverrunCount(SYS_USART_2);
        left = SYS_USART_Available(SYS_USART_2);
        printf("[T1] drop=%u full=%u ovr=%u left=%u\r\n",
               (unsigned)drop, (unsigned)full, (unsigned)ovr, (unsigned)left);

        SYS_USART_RxFlush(SYS_USART_2);
        vTaskDelay(pdMS_TO_TICKS(3000));
    }
}
```

**操作**：串口助手打开上位机那一路（115200），看到 `[T1] reset done...` 后连续发 **1024 个字节**。

**判定**：

| 条件 | 期望 |
|---|---|
| 默认 `SYS_USART_RX_BUF_SIZE` = 64 | `drop` > 0（约 960 上下，取决于中断响应），`full` ≈ `drop`，`left` ≈ 60 多 |
| 把 `FWLIB\inc\sys_usart.h` 里 `SYS_USART2_RX_BUF_SIZE` 改成 **2048** 重新编译 | `drop` = 0，`left` ≈ 1024 |
| 造 ORE | 发得极快时 `ovr` 可能非 0（说明是中断被挡，不是缓冲小）——这条不是必现，记下来即可 |

> 注意：`SYS_USART_RxDropReset()` 只清计数，不清缓冲；测试里先 Reset 再灌数据。

## 三、T2 — A2：日志并发，不再挂起调度器（约 30 分钟）

**目的**：证明 `vTaskSuspendAll()` 换成互斥量之后，**擦扇区/清日志期间整个系统不再停摆**，且写入与补传并发不脏数据。

前置：`W25QXX_Init(0)` 先 `W25QXX_LogInit()`（在你的 main 里加，或在本文件的任务里加一次）。

```c
static volatile uint32_t g_hb_max_ms = 0;
static volatile uint32_t g_write_busy = 0;
static volatile uint32_t g_write_err  = 0;

/* 心跳：每 100ms 一次，记录最大间隔 —— 调度器被挂起就会体现在这个数上 */
static void vTestHeartbeat(void *pv)
{
    TickType_t last = xTaskGetTickCount();
    (void)pv;
    for (;;)
    {
        TickType_t now;
        uint32_t   gap;
        vTaskDelay(pdMS_TO_TICKS(100));
        now = xTaskGetTickCount();
        gap = (uint32_t)((now - last) * portTICK_PERIOD_MS);
        if (gap > g_hb_max_ms) { g_hb_max_ms = gap; }
        last = now;
    }
}

/* 写入：每 20ms 一条，载荷里带序号，便于人工抽查最后一次读到的内容 */
static void vTestLogWriter(void *pv)
{
    uint8_t  buf[8];
    uint32_t ts = 0;
    (void)pv;
    for (;;)
    {
        uint8_t err;
        uint8_t i;
        for (i = 0; i < 8; i++) { buf[i] = (uint8_t)(ts + i); }
        err = W25QXX_LogWrite(ts, buf, 8);
        if (err == W25QXX_LOG_ERR_BUSY) { g_write_busy++; }          /* 清日志期间，正常 */
        else if (err != 0)              { g_write_err++; }           /* 其它错，异常 */
        ts++;
        vTaskDelay(pdMS_TO_TICKS(20));
    }
}

/* 补传/巡检：每 5s 把全库读一遍，检查条数与 CRC */
static void vTestLogReplay(void *pv)
{
    static uint8_t buf[32];
    uint16_t last_n = 0;
    (void)pv;
    for (;;)
    {
        uint16_t n, cap, head, i, bad = 0;
        uint32_t ts;
        uint16_t len;

        W25QXX_LogStat(&n, &cap, &head);
        for (i = 0; i < n; i++)
        {
            if (W25QXX_LogRead(i, &ts, buf, &len) != 0) { bad++; }
        }
        printf("[T2] n=%u cap=%u head=%u lost=%u bad=%u busy=%u err=%u maxgap=%ums %s\r\n",
               (unsigned)n, (unsigned)cap, (unsigned)head,
               (unsigned)W25QXX_LogLost(), (unsigned)bad,
               (unsigned)g_write_busy, (unsigned)g_write_err,
               (unsigned)g_hb_max_ms,
               (n >= last_n) ? "MONO-OK" : "MONO-BAD");
        last_n = n;
        vTaskDelay(pdMS_TO_TICKS(5000));
    }
}

/* 分步清：开机 60s 后清一次，展示"清日志不卡界面" */
static void vTestLogClear(void *pv)
{
    uint8_t done = 0;
    (void)pv;
    vTaskDelay(pdMS_TO_TICKS(60000));
    printf("[T2] clear start\r\n");
    while (!done)
    {
        uint8_t err = W25QXX_LogClearStep(&done);
        if (err != 0) { printf("[T2] clear err=%s\r\n", W25QXX_LogErrStr(err)); break; }
        vTaskDelay(pdMS_TO_TICKS(20));       /* 两次擦除之间让出 CPU */
    }
    printf("[T2] clear done, busy=%u maxgap=%ums\r\n",
           (unsigned)g_write_busy, (unsigned)g_hb_max_ms);
    vTaskDelete(NULL);
}
```

**判定**：

| 观察项 | 期望 | 说明 |
|---|---|---|
| `maxgap` | 全程 < 200ms（稳定后基本 100~110ms） | 改前擦扇区会看到 400ms~1s 的尖峰，这就是本项改动要消灭的现象 |
| `bad` | 恒 0 | 每条都能通过 CRC16 读出来，说明并发写没写坏 |
| `MONO-OK` | 一直是 | `n` 不会倒退（读的同时在写，最多是旧值） |
| `lost` | 只在写满 2048 条之后开始增长 | 20ms 一条 ≈ 41 秒写满，之后覆盖最旧 —— 只增不减是对的 |
| `busy` | 清日志那 1 秒内会出现几条 | 说明写接口在清日志期间返回 BUSY 而不是写坏 |
| `err` | 恒 0 | 非 0 就是真异常（把 `err` 值发我） |

> 想更狠一点：把 writer 的 `vTaskDelay(20)` 改成 `vTaskDelay(1)`，让写入压满，`maxgap` 与 `bad` 仍应达标。

## 四、T3 — B1：ESP8266 多任务并发不打架（约 5 分钟）

**目的**：证明模块内那把递归锁真的把"一条命令一段回复"保护住了。

前置：给模块供电、P10 跳线选 WIFI 侧，先 `ESP8266_Init(115200)`。

```c
static volatile uint16_t g_at_fail = 0;
static volatile uint16_t g_at_ok   = 0;

static void vTestAtStress(void *pv)
{
    uint32_t i;
    (void)pv;
    for (i = 0; i < 100; i++)
    {
        if (ESP8266_TestAT() == 0) { g_at_ok++; }
        else                       { g_at_fail++; }
        vTaskDelay(pdMS_TO_TICKS(5));
    }
    printf("[T3] done ok=%u fail=%u\r\n", (unsigned)g_at_ok, (unsigned)g_at_fail);
    vTaskDelete(NULL);
}
```

创建**两个**这样的任务（各自 100 次），另外再开一个任务每 50ms 调一次 `ESP8266_Flush(); ESP8266_HasAsyncData();` 捣乱：

```c
static void vTestAtChaos(void *pv)
{
    (void)pv;
    for (;;)
    {
        ESP8266_Flush();
        (void)ESP8266_HasAsyncData();
        vTaskDelay(pdMS_TO_TICKS(50));
    }
}
```

**判定**：`ok` = 200、`fail` = 0。出现非 0 或回复串到别的任务（比如某次 `AT` 的回复变成 `+IPD`）就算失败，把日志发我。

> 反证法（可选）：把 `esp8266.c` 里 `SendCmd` 包装的加锁/解锁注释掉重编，同样并发跑，应该能复现 `fail > 0`。

## 五、T4 — B2：MQTT 下行不串包（约 30 分钟）

**目的**：证明 `MQTT_Poll()` 取"长度/指针"的时机正确，长跑不出现乱码或重复。

```c
static void vTestMqttSub(void *pv)
{
    uint32_t got = 0;
    (void)pv;
    for (;;)
    {
        MqttMsg_t m;
        if (MQTT_Poll(&m) == MQTT_OK)
        {
            got++;
            printf("[T4] #%u topic=%s payload=%s alive=%u conn=%u\r\n",
                   (unsigned)got, m.topic, m.payload,
                   (unsigned)MQTT_IsAlive(), (unsigned)MQTT_IsConnected());
        }
        (void)MQTT_KeepAliveService();

        if (MQTT_IsAlive() == 0)
        {
            printf("[T4] link dead, connack=%u\r\n", (unsigned)MQTT_GetConnackCode());
        }
        vTaskDelay(pdMS_TO_TICKS(20));
    }
}
```

前置：`ESP8266_Init` → `ESP8266_SetMode(ESP8266_MODE_STA)` → `ESP8266_JoinAP(...)` → `MQTT_ConnectSimple("broker.emqx.io", "f407-test")`，然后 `MQTT_Subscribe("cmd/f407-test", 0)`。

**操作**：用上位机（MQTTX / mosquitto_pub）向 `cmd/f407-test` 依次发 **50 条** payload 为 `m1` … `m50` 的消息，每条间隔 200ms。

**判定**：

| 观察项 | 期望 |
|---|---|
| 收到条数 | 50 条，不多不少 |
| payload | `m1`~`m50` 各一次，无乱码、无截断、无粘连 |
| `alive` | 一直 1；拔网线/断 AP 后 120 秒内变 0 并打印 `link dead` |
| 长跑 | 让 broker 每 10 秒自动发一条，坚持 30 分钟，`got` 与 sent 一致 |

> 复现旧 bug（可选）：把 `MQTT_Poll()` 里"先探活再取指针"的两句调回原顺序重编，同样发 50 条，应当能观察到 payload 错位或长度对不上。
